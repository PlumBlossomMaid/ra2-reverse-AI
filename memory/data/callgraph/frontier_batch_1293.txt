0x410320 —— COM 样式读取器（87B 全程读完）
================================================================================
签名：thiscall（ecx=this）+ 3 栈参（[esp+4]=对象指针、[esp+8]、
[esp+0xC]=目标缓冲）ret 0xC（HRESULT）。
机器码骨架（0x410320-0x410376）：
    mov esi, [esp+0xC]; test esi,esi; jne 0x410334
    mov eax, 0x80004003; ret 0xC                ; ★对象为空 → E_POINTER
    0x410334:
    mov edi, [esp+0xC]                          ; edi = 目标缓冲
    mov eax, [esi]; push 0; lea ecx,[esp+0x14]
    push 4; push ecx; push esi
    call dword ptr [eax+0x10]                   ; ★虚表槽 0x10 读（4 字节）
    jl 0x410372                                 ; ★HRESULT < 0 → 返回
    mov edx, [edi]; mov ebx, [esi]
    push 0; mov ecx, edi
    call dword ptr [edx+0x30]                   ; ★虚表槽 0x30 获取
    push eax; push edi; push esi
    call dword ptr [ebx+0x10]                   ; ★虚表槽 0x10 二次读
    jl 0x410372
    mov ecx, [esp+0x14]; test ecx,ecx; je 0x410372

    0x410372: ret 0xC

宿主虚表槽 0x10 读 4 字节（HRESULT 门控，负值即返）→ 目标虚表槽
0x30 获取派生值 → 二次虚表槽 0x10 读 → 成功清目标 +0x20 标志字节。
★IStream/ISerializable 形态（Read 虚表族）。3 unique caller（30 callsite）。
