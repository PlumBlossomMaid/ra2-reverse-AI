一、链语义（disasm 全段复核——本批补全 0x7D 系核心身份）
------------------------------------------------
0x7CA530(52B) = MSVC6 CRT vsprintf 薄包装（0x7C-0x7D 静态 CRT 区）：
  push ebp/mov ebp,esp/sub esp,0x20（32B 栈帧）
  mov eax,[ebp+8]                 ; ★ arg1 = 目标缓冲
  mov dword[ebp-0x14],0x49        ; ★ 本地 +0x14 槽 = 0x49（'I'）
  push eax / mov [ebp-0x18],eax / mov [ebp-0x20],eax
  call 0x7D15A0                   ; ★ ★0x7D 系 = _strlen（TRS 017
                                   ;   已登记 + string_length_legacy
                                   ;   REIMPL，交叉实锤）→ eax=

  mov [ebp-0x1C],eax              ; 本地 +0x1C 槽 = strlen(arg1)
  lea eax,[ebp+0x10] / push eax   ; ★ 第三栈参 = &arg3（va_list 槽）
  lea eax,[ebp-0x20] / push [ebp+0xC] / push eax
  call 0x7D170D                   ; ★ ★0x7D170D = vsprintf 内核
                                   ;   （cdecl 3 栈参：&本地结构, 格式
                                   ;   串, va_list）
  add esp,0x10 / leave / ret
  ★本地结构 [ebp-0x20..0x14] = {arg1(dst), arg1, 0x49='I', strlen}
  传给 0x7D170D 作为首参——MSVC6 _vsnprintf/_vsnprintf_s 内部
  FILE/缓冲描述结构的简化形态（'I' 是字符/整数标志，见下）。

★0x7D170D 内核开头（本批补全，0x7D170D-0x7D1752 已读）：
  sub esp,0x1C4（452B 大栈帧）/ push ebx/esi/edi
  mov esi,[ebp+0xC]               ; 格式串
  mov al,[esi] / test al,al / je 0x7D2113（空串早退）
  cmp dword[0x87C79C],1           ; ★★[0x87C79C] = locale 门控
                                   ;  （MSVC6 per-thread locale
                                   ;   flag）→ jle 走 [0x87C590]
                                   ;   （locale 表）快速路径
  movzx eax,al / push 8 / push eax / call 0x7D1311（char 转换/分类）
  —— vsprintf/vsprintf_s 内核形态（格式串逐字符解析 + locale
  敏感分类）实锤。

★同族变体 0x7CA564（本批补全，0x7CA564-0x7CA595 已读）：
  mov dword[ebp-0x14],0x42        ; ★ 常量 0x42（'B'）
  mov dword[ebp-0x1C],0x7FFFFFFF  ; ★ 上限 = INT_MAX（缓冲上限）
  call 0x7D21BE                   ; ★ ★0x7D21BE = **宽**内核
                                    ;  （vswprintf 形态）
  add esp,0xC / dec dword[ebp-0x1C]
  ★0x7D21BE 内核开头（本批补全）：
    sub esp,0x450（1104B 大栈帧）
    mov eax,[ebp+0xC] / add dword[ebp+0xC],2  ; ★★宽字符步进（+2
                                        ; 字节/字符——宽格式串）
    mov bx,word ptr [eax]          ; ★★读 **word**（宽字符）
  —— 与 0x7CA530（窄：0x7D170D 读 byte）形成 窄/宽 双包装对：
  0x7CA530 = vsprintf（'I' + strlen 预存），0x7CA564 = vswprintf
  （'B' + 0x7FFFFFFF 上限 = 无界模式）。'I'/'B' = MSVC6 内部缓冲
  描述字符标志（'B' = buffer 模式带上限 / 'I' = 无限模式），
  推断标注（未 golden 实证）。

★可观测行为总结：3 参 cdecl 薄包装（dst, fmt, ...va）——本地填

   golden 对拍需整套 locale/printf 状态，投入产出失衡；batch 169
   已确认 semantics，本批补全核心身份与窄/宽确认，达归档级。
