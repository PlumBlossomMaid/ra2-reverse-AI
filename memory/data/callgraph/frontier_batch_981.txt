一、链语义（disasm 全 92B 复核 + batch 049 交叉）
------------------------------------------------
0x7CB1F9(92B) = MSVC6 CRT 底层流缓冲 flush（_flush 内部，
0x7C-0x7D 静态 CRT 区）——cdecl 1 栈参（FILE* 流槽）：
  push ebx / push esi / push edi
  mov esi,[esp+0xC]              ; ★参数=FILE* 流槽
  xor ebx,ebx                    ; ★ebx=0（返回值预置 0 成功）
  mov eax,[esi+0xC]              ; ★标志字
  mov ecx,eax / and ecx,3 / cmp cl,2   ; ★低 2 位==2
  jne 0x7CB246                   ; 非 2 → 跳过 flush
  test ax,0x108                  ; ★0x108=_IOFLRTN(0x10)|_IOBINARY(0x8)
  je 0x7CB246                    ; 无带缓冲+二进制组合 → 跳过
  mov eax,[esi+8] / mov edi,[esi] / sub edi,eax   ; ★edi=游标-缓冲基址
                                  ; =待写字节数
  test edi,edi / jle 0x7CB246    ; ★无待写 → 跳过
  push edi / push eax / push [esi+0x10]（★句柄）
  call 0x7D0D55                  ; ★★缓冲写设施（batch 046/971 交叉）
  add esp,0xC
  cmp eax,edi / jne 0x7CB23F     ; ★写回数 != 待写数 → 失败
  mov eax,[esi+0xC] / test al,0x80（★EOF 标记位7）/ je 0x7CB246
  and al,0xFD                    ; ★成功清 _IOFLRTN（~0x02）
  mov [esi+0xC],eax / jmp 0x7CB246
  0x7CB23F: or dword [esi+0xC],0x20（★失败置 _IOERROR）/ or ebx,0xFFFFFFFF
                                  ; ★返 -1
  0x7CB246（复位）：
  mov eax,[esi+8]                ; ★缓冲基址
  and dword [esi+4],0            ; ★清 +0x4
  mov [esi],eax                  ; ★游标=缓冲基址
  pop edi / mov eax,ebx / pop esi / pop ebx / ret
★语义（batch 049 名准）：**底层流缓冲 flush**——仅当流标志低 2 位
==2 且 0x108（_IOFLRTN|_IOBINARY）置位且有待写数据（游标-基址>0）
时调 0x7D0D55 缓冲写；写回数与请求数一致且 EOF 标记（0x80）置位时
清 _IOFLRTN（0x02），失败置 _IOERROR（0x20）并返 -1；无论是否刷
新，收尾复位游标=缓冲基址、清 +0x4、返 ebx（0 成功/-1 失败）。
★原版怪癖（batch 049 已录，照抄）：
  ① 三重条件门（低 2 位==2 + 0x108 + 待写>0）全满足才刷；
  ② 成功清 _IOFLRTN（0xFD 掩码）但**保留 EOF 标记**（0x80 只读
     不参与清除——位7 标记 EOF 状态独立于 flush）；
  ③ 失败置 _IOERROR（0x20）+ 返 -1，但**仍执行复位**（游标=基址、
     清 +0x4）——失败路径不停在中间态；
  ④ 复位无条件执行（0x7CB246 三处汇合——跳过/成功/失败都复位）；
  ⑤ 待写字节数 = 游标([esi]) - 缓冲基址([esi+8])——FILE 内部游标
     前移形态。
