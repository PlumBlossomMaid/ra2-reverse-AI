0x643C50  x87 浮点累加除法（9 callers，0x643C50-0x643CAD 前段读完）
================================================================================
机器码（前 0x5D）：
    push ebp / mov ebp,esp / and esp,0xfffffff8 / sub esp,0xC
    push esi / mov esi,ecx
    fld qword[0x7E2800]             ; ★ 初始 0.0（batch 186 的 [0x7E2800] 同常量）
    movsx ecx,byte[esi+0x61]       ; ★ this->+0x61 字节符号扩展 = 计数
    test ecx / jle 跳过累加
  累加:
    lea eax,[esi+8] / fadd qword[eax] / add eax,8 / dec ecx / jne 累加
    ; ★ 累加 +0x08 起 N 个 double（fild 给的 N）
  跳过累加:
    fild dword[esp+4]              ; ★ (double)N
    fdivr st(1)                    ; ★ ST(1) = 累加 / N（fdivr 反向除）
    fdiv qword[esi+0x48]           ; ★ / this->+0x48（除数）
    fmul qword[0x7E2AC0]           ; ★ × [0x7E2AC0] 系数
    call 0x7C5F00                  ; ★ _ftol 截断（batch 083/092 同族）
    fstp st(0)
    mov ecx,[ebp+8] / mov edx,eax
    fld qword[esi+ecx*8+8]         ; ★ 取第 ecx 项 double
    fld qword[esi+0x48] / fmul qword[0x7E3808] / fmul qword[ebp+0xC]
    fst qword[esi+ecx*8+8]         ; ★ 第 ecx 项 *= (+0x48×0x0C)
    fcomp qword[esi+0x48] / fnstsw ax / test ah,0x41  ; ★ 与 +0x48 比较
    ; ★ fcomp + test ah,0x41（C3|C0）——batch 166 的 0x750010 同款判定

语义：**x87 浮点均值+比例缩放（数值计算链）**——从 +0x08 起按 +0x61 计数
累加 N 个 double，fild(N) + fdivr 求均值，再除 this->+0x48 并乘
[0x7E2AC0]，_ftol 截断成 int；随后把第 ecx 项乘 (+0x48×[0x7E3808]=0.01)
写回，最后 fcomp + test ah,0x41 做范围判定。
★**batch 122 的 FPU 精度链延续**：fild/fadd/fdiv/fmul 混合 + _ftol 截断，
常量 [0x7E3808]=0.01（batch 166 实锤）、[0x7E2800]=0.0（batch 186 见过）、
[0x7E2AC0] 新常量。

⚠️ partial：[0x7E2AC0] 系数值（未读字节）、+0x61/+0x48 字段语义、
test ah,0x41 之后的分支（0x643CAD 起）。

0x7CD680  大小写不敏感比较（_stricmp 内联版；9 callers，0x7CD680-0x7CD6D9 读完）
================================================================================
机器码（前 0x59）：
    push ebp / mov ebp,esp / push edi/esi/ebx
    mov ecx,[ebp+0x10]             ; ★ 参数 3：长度 N
    test ecx / je 0x7CD77A         ; ★ N==0 → 直接走尾判
    mov esi,[ebp+8] / mov edi,[ebp+0xC]  ; ★ 参数 1/2：两串
    lea eax,[0xB78298] / cmp dword[eax+8],0 / jne 0x7CD6F1
    ; ★ 全局 [0xB78298]+8 一次性初始化门控（locale 缓存推断）
    mov bh,0x41 / mov bl,0x5A / mov dh,0x20   ; ★ 'A'=0x41 'Z'=0x5A ' '=0x20
  循环:
    mov ah,byte[esi] / or ah,ah / mov al,byte[edi]
    test ah / je 0x7CD6D5
    test al / je 0x7CD6D5
    inc esi / inc edi
    cmp ah,bh / jb 0x7CD6C4        ; ★ < 'A' 不转
    cmp ah,bl / ja 0x7CD6C4        ; ★ > 'Z' 不转
    add ah,dh                      ; ★★ 大写字母 +0x20 转小写
   0x7CD6C4:
    cmp al,bh / jb 0x7CD6CE / cmp al,bl / ja 0x7CD6CE / add al,dh
   0x7CD6CE:
    cmp ah,al / jne 0x7CD6DF        ; ★ 转小写后逐字节比较
    dec ecx / jne 循环
   0x7CD6D5: xor ecx,ecx
   0x7CD6D7: cmp ah,al / je 0x7CD77A ; ★ 尾判：首字节相等 → 相等
    ; （0x7CD6DF 起：不等的收尾分支）

语义：**大小写不敏感比较（_stricmp 内联版）**——按长度 N 逐字节比较两串，
每个字节若落在 ['A','Z']（0x41-0x5A）则 `add al,0x20` 转小写后再比；
循环结束再判尾字节。首次调用受全局 [0xB78298]+8 门控（locale 表初始化
推断）。
★0x7C-0x7D MSVC6 静态 CRT 区（项目记忆 ra2_code_region_layout_crt_provenance）——
**与 batch 001 的 0x7C8D20（_stricmp/strcmp）区分**：本条是 CRT 内部的
原地小写化内联实现，不是公开入口。
★注意与 batch 180 的 0x7B5B80（**转大写** and 0xDF）方向相反：本条
**转小写** add 0x20——同族两个方向的规范化实现。

⚠️ partial：[0xB78298] 全局语义（locale 表推断）、0x7CD6DF/0x7CD77A 两
收尾分支（相等/不等返回值约定未读全）、N 参数是否为 strlen 结果。
