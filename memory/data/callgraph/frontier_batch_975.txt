真义（黄金实证 + 机器码全段确认）：0x7C9380 = MSVC6 CRT __aulldiv
（64 位无符号除法）——cdecl 4 dword 栈参 (a_lo, a_hi, b_lo, b_hi)，
返回 edx:eax 64 位商 = (uint64)a / (uint64)b（无符号截断，余数丢弃），
ret 0x10 弹 16 字节。与 batch 963 __allmul（64 位乘法）构成除乘配对。
  机器码（0x7C9380-0x7C93E5，104B 全段）：

    mov eax,[esp+0x18]=b_hi → or eax,eax → jne 0x7C93A2（★b_hi!=0 →
      硬路径）
    快速路径（0x7C938A，b_hi==0）：
      mov ecx,[esp+0x14]=b_lo
      mov eax,[esp+0x10]=a_hi → xor edx,edx → div ecx
        （★edx:eax = 0:a_hi / b_lo → eax=高商，edx=余）
      mov ebx,eax → mov eax,[esp+0xC]=a_lo → div ecx
        （★edx:eax = 余:a_lo / b_lo → eax=低商）
      mov edx,ebx → jmp 0x7C93E3（★edx:高商 + eax:低商）
    硬路径（0x7C93A2，b_hi!=0）：
      mov ecx,eax（=b_hi）→ mov ebx,[esp+0x14]=b_lo
      mov edx,[esp+0x10]=a_hi → mov eax,[esp+0xC]=a_lo
      归一化循环（0x7C93B0）：
        shr ecx,1 / rcr ebx,1（★b_hi:b_lo 联合右移，CF 传位）
        shr edx,1 / rcr eax,1（★a_hi:a_lo 联合右移）

      div ebx（★edx:eax = a 归一化 / b_lo' → eax=预商 esi）
      mov esi,eax → mul dword [esp+0x18]（★esi*b_hi → edx:eax=H）
      mov ecx,eax（★H 低 32——★★H 高 32 被丢弃，原版不检查！）
      mov eax,[esp+0x14]=b_lo → mul esi（★edx:eax = b_lo*esi=L）
      add edx,ecx（★积低64 高32 = L_hi + H_lo，CF=进位）

★原版怪癖（golden 实证，照抄）：
  ① 快速/硬路径分岔在 b_hi；
  ② 快速路径两次 div 分步：先 a_hi/b_lo 得高商（edx 余数接力），再
     (余<<32|a_lo)/b_lo 得低商——★余<b_lo ⇒ 被除数 < b_lo*2^32 ⇒
     低商永不溢出（原版契约，x86 div #DE 防护隐含）；
  ③ 硬路径归一化循环联合右移（shr/rcr 配对——b 最高位被挪进 b_lo
     高位 ⇒ b_lo' ≥ 2^31 使预商不溢出）；
  ④ ★乘法校正只检查 CF（L_hi+H_lo 进位）+ 积低 64 vs a——**H 高 32
     （esi*b_hi 溢出位）被丢弃不检查**（原版怪癖：归一化保证预商至
     多偏大 1，CF 或 64 位比较足以校正；golden 全 1×全 1 等用例经
     此路径实证）；
  ⑤ 返回 edx:eax 64 位商、余数丢弃（收尾 xor edx,edx 高 32 恒 0）；
  ⑥ 除零（b==0）走 div #DE 异常——调用方契约保证非零（golden 不
     含除零用例）。
