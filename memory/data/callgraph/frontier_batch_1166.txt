一、链语义（disasm 全 84B + 三依赖交叉 + 调用点宿主，
0x7DB49A-0x7DB4ED）
--------------------------------------------------------------
签名（推断）：cdecl 5 参（[ebp+8]=目标缓冲、[ebp+0xC]=
double 值、[ebp+0x14]=处理器/回调），返回经 FPU 栈传递的
double / 错误码路径。

主体（0x7DB49A-0x7DB4ED）：
007db49a  push ebp/mov ebp,esp
007db49d  cmp [0x87d700],0                     ; ★★FPE 门控全局（batch
                                              ;   1141 0x7DB54E 同款）
007db4a4  jne 0x7db4cf                        ; 门开 → 错误路径
; → 正常路径（0x7DB4A6，门关）：
007db4a6  push [ebp+0x14]                      ; 参5
007db4a9  fld qword[ebp+0xC]/push ecx/push ecx/fstp [esp]  ; 压 double
007db4b1  push ecx/push ecx/fldz/fstp [esp]    ; 压 0.0
007db4b8  fld qword[ebp+0xC]/push ecx/push ecx/fstp [esp]  ; 再压 double
007db4c0  push [ebp+8]                         ; 参1
007db4c3  push 1                               ; ★固定参 1
007db4c5  call 0x7dbb55                        ; ★★batch 531 宽字符浮点
                                              ;   解析器（FPU 返 double）
007db4ca  add esp,0x24/pop ebp/ret
; → 错误路径（0x7DB4CF，门开）：
007db4cf  call 0x7d168e                        ; ★★batch 034 per-thread
                                              ;   errno +8
007db4d4  push 0xffff
007db4d9  mov dword[eax],0x21                  ; ★errno=0x21（EDOM）
007db4df  push [ebp+0x14]
007db4e2  call 0x7dbc77                        ; ★★batch 192 _controlfp
                                              ;   （FPU 控制字设置）
007db4e7  fld qword[ebp+0xC]/pop ecx/pop ecx/pop ebp/ret
                                              ; ★返 double 兜底
