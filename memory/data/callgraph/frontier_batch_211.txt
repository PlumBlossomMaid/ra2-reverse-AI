0x478720  深拷贝构造（8 callers，全 0x6B 读完）
================================================================================
机器码（全 0x6B，body=108）：
    push esi/edi / mov edi,[esp+0xc]     ; ★ 参数 = 源对象指针
    mov esi,ecx / cmp esi,edi / je 0x478785  ; ★ 自我拷贝 → 直接返回
    mov eax,[esi] / call [eax+0xc]       ; ★ 虚表槽 0xC（清零）
    mov eax,[edi+8] / test / mov [esi+8],eax  ; ★ 拷贝 +8 计数
    je 0x47877A                          ; ★ 计数 0 → +4=NULL、+0xD=0
    lea ecx,[eax*4] / push ecx / call 0x7C8E17  ; ★★ operator new（batch 001）×4
    mov [esi+4],eax / test / je 0x478785 ; ★ 分配失败 → 返回
    mov ecx,[esi+8] / xor eax / test / mov byte[esi+0xd],1  ; ★ +0xD=1 已分配
    jle 0x478785
  拷贝循环:
    mov edx,[edi+4] / mov ecx,[esi+4]
    mov edx,[edx+eax*4] / mov [ecx+eax*4],edx  ; ★ 源数组 → this 数组逐项
    mov ecx,[esi+8] / inc eax / cmp eax,ecx / jl 拷贝循环
    mov eax,esi / pop edi/esi / ret 4
  0x47877A:
    mov [esi+4],0 / mov byte[esi+0xd],0  ; ★ 计数 0 → 空容器

语义：**深拷贝构造/赋值（0x478xxx 区）**——源计数 +8 拷贝 → operator
new（0x7C8E17，×4）分配 +4 数组 → 逐项拷贝源 [edi+4] 数组；自我拷贝
跳过；计数 0 走空容器分支（+4=NULL）；+0xD 置已分配标志。
★**与 batch 201 的 0x4779E0（数组复制构造器，虚表 A→B + 深拷贝）同款
模板**——同 0x47 区、同 +4/+8/+0xC/+0xD 布局、同 operator new ×4 分配。
★虚表槽 0xC（清零/重置）为本布局的清理入口（拷贝前先清）。

⚠️ partial：宿主类（0x478xxx 区，batch 201 的 0x4779E0 邻域）、虚表槽
0xC 身份、自我拷贝跳过的调用方依赖。

0x50EA00  容器释放器（8 callers，全 0x26 读完）
================================================================================
机器码（全 0x26，body=39）：
    push ebx/esi / mov esi,ecx
    mov eax,[esi+4] / cmp eax,ebx / je 0x50EA1E  ; ★ +4 为空 → 直接清
    cmp byte[esi+0xd],bl / je 0x50EA1E   ; ★ +0xD 未置已分配 → 跳过释放
    push eax / call 0x7C8B3D             ; ★★ CRT 释放（batch 197 同款）
    add esp,4 / mov [esi+4],ebx          ; ★ +4 = NULL
    mov byte[esi+0xd],bl                 ; ★ +0xD = 0
    mov [esi+8],ebx                      ; ★ +8 计数 = 0
    pop esi/ebx / ret

语义：**容器释放器（0x50EAxx 区）**——+4 缓冲非空且 +0xD 已分配标志
置位时经 0x7C8B3D（CRT 释放）释放，随后清 +4/+0xD/+8（三字段归零）。
★★**batch 207 的 0x40BD00 构造对**：0x40BD00（缓冲容器构造：+4 缓冲、
+8 计数、+0xC=1、+0xD 已分配、虚表 0x7E180C）的释放侧——构造/释放
同布局闭合（+4/+8/+0xC/+0xD 四字段契约完整）。
★0x7C8B3D（CRT 释放）batch 197 的 0x5C6120、batch 202 的 0x7B4ED0
同款——释放侧第 N 处登记。

⚠️ partial：宿主类（0x50EAxx 区）、+0xC 字段（0x40BD00 置 1 但释放侧
未触碰）、与 0x40BD00 的调用方绑定关系。

0x5253B0  动态数组 Add（8 callers，全 0x4E 读完）
================================================================================
机器码（全 0x4E，body=80）：
    push esi / mov esi,ecx
    mov ecx,[esi+8] / mov eax,[esi+0x10] ; ★ +8 计数 vs +0x10 容量
    cmp eax,ecx / jl 0x5253E5           ; ★ 容量足够 → 直接写槽
    mov al,[esi+0xd] / test al / jne 0x5253C8  ; ★ 已分配标志
    test ecx / jne 0x5253DF             ; ★ 计数非 0 → 走扩容
    mov eax,[esi+0x14] / test eax / jle 0x5253DF  ; ★ 增量 ≤0 → 失败
    mov edx,[esi] / add eax,ecx / push 0 / push eax
    mov ecx,esi / call [edx+8]          ; ★ 虚表槽 8（扩容，新容量=计数+增量）
    test al / jne 0x5253E5             ; ★ 扩容成功 → 写槽
    xor al,al / pop esi / ret 4         ; ★ 扩容失败 → 返 0
  0x5253E5:
    mov eax,[esi+0x10] / mov edx,[esi+4]
    lea ecx,[eax+1] / mov [esi+0x10],ecx ; ★ 容量++（或虚表槽 8 已更新）
    mov ecx,[esp+8] / mov ecx,[ecx] / mov [edx+eax*4],ecx  ; ★ 写槽（值取参数所指）
    mov al,1 / pop esi / ret 4          ; ★ 成功返 1
  紧邻 0x525400（同族遍历）:
    mov edx,[ecx+0x10] / xor eax / test edx / jle 收尾 ...
    ; ★ 按 +0x10 容量遍历（同布局第二成员）

语义：**动态数组 Add（0x525xxx 区）**——向 +4 缓冲追加一项（值取参数
指针所指）：容量 +0x10 足够直接写槽；容量不足时按 +0x14 增量经虚表
槽 8 扩容（新容量 = 旧容量+增量），失败返 0；成功写槽（新项存参数值）
返 1。紧邻 0x525400 是同布局遍历成员。
★★**三件套容器族闭合**：0x478720（深拷贝）/ 0x50EA00（释放）/
0x5253B0（Add）+ batch 207 的 0x40BD00（构造）——**同一 +4 缓冲 +
+8 计数 +0xD 标志 +0x10 容量 +0x14 增量布局的容器五个操作**分三批登记
完毕（0x47/0x50/0x52 三个区各一件，但布局逐字段同构）。
★布局总结：+0 虚表、+4 缓冲指针、+8 计数、+0xC 标志（=1）、+0xD
已分配标志、+0x10 容量、+0x14 增量——跨三区同构容器族。

⚠️ partial：宿主类（0x525xxx 区）、虚表槽 8（扩容）身份、+0x14 增量
语义、值取参数所指的调用方契约、紧邻 0x525400 身份。
