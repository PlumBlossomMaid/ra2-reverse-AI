================================================================================
0x7CA5D3  CRT::wcscmp（23 callers）
================================================================================
身份实锤：YRpp CRT.h:27-28
    static int __cdecl wcscmp(const wchar_t *Str1, const wchar_t *Str2)
        { JMP_STD(0x7CA5D3); }
地址 + `__cdecl` 两宽串参 + 尾段 -1/0/1 归一三证吻合。

机器码（全 0x35B）：
    mov edx, [esp+4]              ; Str1
    mov esi, [esp+0xC]            ; Str2
    mov cx, [esi]                 ; ★ 2 字节/次取宽字符
    movzx eax, word [edx]
    movzx edi, cx
    sub eax, edi / jne 收尾       ; 不等 → 直接出结果
    test cx, cx / je 收尾          ; 等且遇 '\0' → 0
    inc edx×2 / inc esi×2 / jmp 回环
  收尾:
    test eax / jge / or eax, -1    ; 负数 → -1
    jle 返 0 / push 1 / pop eax    ; 正数 → +1
    ret

语义：**标准 CRT 宽字符字典序比较**——逐 wchar_t 比较，遇差或 '\0' 停止，
结果归一到 -1/0/1。0x7C-0x7D 静态 CRT 层（项目既有实证
ra2_code_region_layout_crt_provenance）的宽字符家族成员：与 batch 098 登记的
CRT::wcsncpy（0x7CA422）、batch 093 的 strcmpi（0x7C8D20）同源，本批补上
**比较三件套的最后一块（wcscmp）**——wcsncpy 拷贝 / wcscmp 比较 /
wcslen（batch 097 已见 0x7CA405）齐活。

⚠️ partial：无（全 0x35B 读完）。返回值归一细节（`test eax/jge/or eax,-1`
的负数分支）与 MSVC6 wcscmp 标准实现一致。

================================================================================
0x7CA650  _alloca_probe（栈探测）（22 callers）
================================================================================
机器码（全 0x2F）：
    push ecx                       ; ★ 备份返回地址经由 ecx 保存
    cmp eax, 0x1000
    lea ecx, [esp+8]
    jb 收尾
  probe_loop:
    sub ecx, 0x1000                ; 每 0x1000 页触碰一次
    sub eax, 0x1000
    test dword [ecx], eax          ; ★ 写探测（防跨越守卫页）
    cmp eax, 0x1000 / jae probe_loop
  收尾:
    sub ecx, eax
    mov eax, esp
    test dword [ecx], eax          ; 末段探测
    mov esp, ecx                   ; ★ 栈指针调整 = 分配
    mov ecx, [eax]                 ; 取回调用方 ecx
    mov eax, [eax+4]               ; 取回调用方 eax
    push eax / ret

语义：**MSVC6 CRT 的栈探测函数（_alloca_probe / __chkstk 类）**——在动态栈
分配（alloca）前逐 0x1000 字节触碰新栈页，防止大分配一次跳过守卫页触发
异常；最后 mov esp 完成分配并还原寄存器。**引擎在每次大栈帧函数（如 batch
109 的 0x56EB80、batch 110 的 0x5E2700 0x800 栈帧）前经 CRT 走它**——这是
「为什么原版函数能开 0x800/0x468 栈帧不炸」的机制答案。

⚠️ partial：具体导出名（_alloca_probe 与 __chkstk 的区分未逐一比对 MSVC6
CRT 符号表，形态已实锤为「0x1000 页步进探测」标准实现）。

0x5E2700  对话框控件刷新链（22 callers，0x86B 全读完）
================================================================================
YRpp 无覆盖按机器码判类。调用方含 0x5E2790（同构第二成员，仅常量不同
0x53F vs 0x5B5）。

机器码还原（主结构）：
    sub esp, 0x800                 ; ★ 0x800 栈帧
    mov ecx, 0x200 / xor eax, eax
    lea edx, [esp]
    rep stosd                       ; ★ 0x200 dword 清零（0x800B 局部）
    mov ecx, [esp+0x810]           ; 参数
    lea eax, [esp+0x814]           ; 局部
    push eax / push ecx / push edx
    call 0x7CA858                  ; ★ CRT（0x20 栈帧标准函数，疑 strncpy 系）
    add esp, 0xC
    call 0x775B10                  ; ★ = 读 [0xB72F44]（batch 094 实锤：
                                   ;   UI 对话框栈「当前 hDlg」全局）
    test eax / je 收尾
    call 0x775B10                  ; 再读（确认非空）
    mov edi, [0x7E14A8]            ; ★ GetDlgItem（batch 099 实锤导入槽）
    mov esi, eax
    push 0x53F / push esi
    call edi                       ; ★ GetDlgItem(hDlg, 0x53F) 取控件
    test eax / jne 已得
    mov ecx, esi
    call 0x7759B0                  ; ★ 替代路径（hDlg 解析）
    mov esi, eax
    push 0x53F / push esi / call edi
    test eax / je 回环             ; 取不到 → 循环重试
  已得:
    mov ecx, [esp+0x80C] / push eax / lea edx, [esp+0xC]
    call 0x5E2820                 ; ★ 尾部组装：wcslen(0x7CA405) 开头，字符串写入
    ret

语义：**「拿到当前对话框的某控件并更新其文本」的驱动链**——大栈帧清零 →
CRT 0x7CA858 处理参数 → 经 [0x775B10]=[0xB72F44]（batch 094 当前 hDlg）两次
读取 → GetDlgItem(hDlg, 0x53F) 取控件 → 取不到走 0x7759B0 替代解析 → 尾部
0x5E2820 用 wcslen（batch 097）组装字符串。**这是 batch 094/099 UI 家族
（BeginDialog/EndDialog/消息泵）的控件级消费者**——三条已登记证据
（0xB72F44 栈全局、[0x7E14A8] GetDlgItem、0x7CA405 wcslen）在本函数咬合成链。

⚠️ partial：0x7CA858 的确切 CRT 身份（疑 strncpy/strcpy 系未实锤）、0x7759B0
替代路径、控件 ID 0x53F/0x5B5 对应的对话框控件语义。
