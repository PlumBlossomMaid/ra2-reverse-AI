0x5C6370 —— 字节流解码展开器（126B 全程读完）

签名：thiscall（ecx=目标缓冲起始）+ 1 栈参（[esp+0x14]=上限长度）ret 4。
机器码（0x5C6370-0x5C63ED）：

    mov al, [edx]             ; al = 字节流当前字节
    inc edx                   ; 游标前进

    and eax, 0xFF; mov edi, eax; jmp 0x5C63AA

    mov cl, [edx]; inc edx
    cmp eax, 1; mov edi, ecx; jle 0x5C63AB   ; 前缀 1 → 1 字节值
    mov al, [edx]; shl edi, 8; or edi, eax   ; 前缀 2 → ★2 字节值（大端）
    0x5C63AA: inc edx
    0x5C63AB:
    mov ecx, [esp+0x14]; lea esi, [ecx*4]    ; esi = 上限×4
    cmp edi, esi; ja 0x5C63E7                ; ★长度>上限 → 跳过
    mov al, [edx]; mov ebx, edx; and al, 0x80
    neg al; sbb al, al; and eax, 0xFF         ; ★填充字节 = bit7 ? 0xFF : 0
    mov edx, esi; sub edx, edi                ; 填充数量 = 上限-长度

    test edx, edx; jle 0x5C63DA

    0x5C63DA:
    test edi, edi; jle 0x5C63E7
    0x5C63DE: mov dl, [ebx]; dec ecx; inc ebx; dec edi; mov [ecx], dl; jne 0x5C63DE  ; ★反向拷贝数据
