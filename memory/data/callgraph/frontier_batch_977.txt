区）——cdecl 1 栈参（FILE* 流槽），流锁包裹单字符/串写：
  push ebx / push esi
  mov esi,[esp+0xC]                 ; ★参数=FILE* 流槽
  push edi
  push esi / call 0x7CB3F9          ; ★★流锁 Enter（batch 966 归档）
  push esi / call 0x7D2B7F          ; ★★FILE 设施（定位/条件检查）
  mov edi,eax
  lea eax,[esp+0x20]                ; ★栈临时槽（宽/窄转换缓冲）

  push [esp+0x20]                   ; ★再推参数
  push esi

                                    ;  后邻——batch 971 交叉）

  call 0x7D2C0C                     ; ★★FILE 设施（flush/计数）
  push esi / call 0x7CB44B          ; ★★流锁 Leave（batch 961 归档）
  add esp,0x20                      ; cdecl 弹 8 参
  mov eax,ebx / pop 族 / ret
