语义：**-2 特判 getter（族读侧）**——对 +0x63/+0x67 对（batch 197 的
0x697020 setter 的字段对）：
- 正常态（setter 写过常规值）：(+0x63=v, +0x67=-1) → 返 +0x63=v；
- -2 特判态（setter 收到参数 -2）：(+0x63=-1, +0x67=-2) → 返 -2；
- 其他组合（如 +0x67==-2 但 +0x63!=-1）→ 返 +0x63。
★★**闭合 batch 194/197/198 一直标 partial 的「-2 特判含义」**：这个族是
**带哨兵编码的 int 值对**——值域扩展一个「-2」哨兵：setter 收 -2 时写
(-1,-2) 特殊组合，getter 读回 -2；正常值 v 写 (v,-1)。getter 的存在把
族语义从「两个并排 setter」升级为「**int 字段的哨兵编码存取对**」——
-2 作为「未设置/特殊值」贯穿读侧写侧。
★地址 0x696F70 在首份 setter 0x696F90 之前——编译器按符号序布局，非
逻辑序。

⚠️ partial：字段业务语义（会话/房间属性推断）、-2 哨兵的业务含义、族

0x4779E0  数组复制构造器（8 callers，全 0x86 读完）
================================================================================
机器码（全 0x86，body=136）：
    push ebx/esi/edi
    mov edi,[esp+0x10]           ; ★ 参数 = 源对象指针
    mov esi,ecx                  ; ★ this
    xor ebx,ebx / cmp esi,edi
    mov [esi+4],ebx / mov [esi+8],ebx      ; ★ 清 +4/+8
    mov byte[esi+0xc],1 / mov byte[esi+0xd],bl  ; ★ +0xC=1、+0xD=0
    mov [esi],0x7E4DB8           ; ★ 装虚表 A（0x7E4DB8）
    je 0x477A4E / call [0x7E4DC4]  ; ★ 绝对间接调用（[0x7E4DC4]=虚表 0x7E4DB8
                                    ;   槽 +0xC 提前解析的函数指针——非导入槽，
                                    ;   是虚表内函数指针的数据直调）
    mov eax,[edi+8] / cmp eax,ebx / mov [esi+8],eax  ; ★ 拷贝 +8 计数
    je 0x477A48 / shl eax,2 / push eax
    call 0x7C8E17                ; ★★ operator new（batch 001 已登记，×4）
    cmp eax,ebx / mov [esi+4],eax / je 0x477A4E
    mov ecx,[esi+8] / xor eax,eax / cmp ecx,ebx
    mov byte[esi+0xd],1 / jle 0x477A4E
  复制循环:
    mov ecx,[edi+4] / mov edx,[esi+4] / mov ecx,[ecx+eax*4]
    mov [edx+eax*4],ecx          ; ★ 源 +4 数组项 → 本 +4 数组（逐项拷贝）
    mov ecx,[esi+8] / inc eax / cmp eax,ecx / jl 复制循环
    jmp 0x477A4E
  0x477A48:
    mov [esi+4],ebx / mov byte[esi+0xd],bl   ; ★ 计数 0 → +4=NULL
  0x477A4E:
    mov edx,[edi+0x10] / mov [esi+0x10],edx  ; ★ 拷贝 +0x10
    mov eax,[edi+0x14] / mov [esi+0x14],eax  ; ★ 拷贝 +0x14
    mov [esi],0x7E4E78           ; ★ 换虚表 B（0x7E4E78）
    mov eax,esi / pop edi/esi/ebx / ret 4

语义：**数组复制构造器（0x477xxx 区）**——this+4=动态数组指针、+8=计数：
清 +4/+8，+0xC=1、+0xD=0 标志，装虚表 A（0x7E4DB8）后经其槽 +0xC（数据直调
[0x7E4DC4]）完成基类部分；从源拷贝 +8 计数，非 0 则 operator new（0x7C8E17，
计数×4）分配 +4 数组并逐项拷贝（源 [edi+4] → this [esi+4]），+0xD 置 1
（已分配标志推断）；+0x10/+0x14 两字段直接拷贝；最后换虚表 B（0x7E4E78）
收尾——**基类构造 + 动态数组深拷贝 + 虚表切换**的完整复制构造。
★[0x7E4DC4] 非导入槽：IAT 解析未命中，是虚表 0x7E4DB8 内函数指针的
绝对直调（0x7E4DB8+0xC 槽位）——虚表槽提前解析的数据调用模式。
★operator new（0x7C8E17）batch 001 已登记——本批第三条调用该 CRT 分配器
（batch 199 的 0x7B6290 也是）。

⚠️ partial：宿主类（0x477xxx 区）、虚表 A/B（0x7E4DB8/0x7E4E78）身份、
