0x6C9890  链表节点重连（10 callers，0x6C9890-0x6C98D0 读完）
================================================================================
机器码（全 0x40）：
    push esi / mov esi,[esp+8]   ; ★ 参数：新节点
    push edi / mov edi,ecx       ; ★ this（宿主）
    cmp [edi+4],esi / je 收尾    ; ★ 已是目标 → 直接返回
    test esi / je 0x6C98B6      ; ★ 新节点空 → 只清旧
    mov ecx,[esi+8] / test / je 0x6C98B6  ; ★ 新节点已挂别处（+8=宿主）
    mov eax,[ecx] / push 0 / call [eax+4] ; ★ 虚表槽 0x04（旧宿主解除通知）
    mov [esi+8],0
  0x6C98B6:
    mov eax,[edi+4] / test / je 0x6C98C4 ; ★ 旧节点
    mov [eax+8],0                ; ★ 旧节点 +8 宿主清 0
  0x6C98C4:
    mov [edi+4],esi              ; ★ this->+4 = 新节点
    test esi / je 收尾
    mov [esi+8],edi              ; ★ 新节点 +8 = this（双向链接）
  收尾: pop edi / pop esi / ret 4

语义：**链表节点重连（父子链接 setter）**——把 this->+4 指向的节点换成
新节点：旧节点 +8 宿主清 0、新节点若已挂别处先经虚表槽 0x04 解除再挂到
this（新节点 +8=this），双向链保持一致。
★0x6Cxxxx 区（batch 179 的 0x6CB920 超级武器族同区）——对象父子链管理。

⚠️ partial：宿主类、虚表槽 0x04（解除通知）身份、+4/+8 字段名。
