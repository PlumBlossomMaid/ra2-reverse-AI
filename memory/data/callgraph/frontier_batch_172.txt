0x5D1C20  向量追加（Add_Item 形态；11 callers，0x5D1C20-0x5D1C7F 读完）
================================================================================
机器码（0x5F 读完）：
    mov eax,[esp+4]              ; ★ 参数：待追加项
    test eax / je 返 -1          ; ★ 空项 → -1
    mov eax,[ecx+0x28]           ; ★ this->+0x28 上限/容量
    lea esi,[ecx+0x18]           ; ★ esi = this->+0x18（vector 子结构）
    mov ecx,[ecx+0x20]           ; ★ this->+0x20 计数
    cmp eax,ecx / jl 直接追加    ; ★ 容量 >= 计数 → 跳过扩展
    mov al,[esi+0xD] / test / jne 扩展
    test ecx / jne 返 -1
  扩展:
    mov eax,[esi+0x14] / test / jle 返 -1
    mov edx,[esi] / add eax,ecx  ; ★ 新容量 = 计数 + 步长
    push 0 / push eax / mov ecx,esi / call [edx+8]  ; ★ 虚表槽 8：Resize
    test al / je 返 -1
  直接追加:
    mov eax,[esi+0x10]           ; ★ +0x10 计数（此子结构）
    mov edx,[esi+4]              ; ★ +0x4 缓冲
    lea ecx,[eax+1] / mov [esi+0x10],ecx   ; ★ 计数++
    mov ecx,[esp+8] / mov [edx+eax*4],ecx  ; ★ 4B/项写缓冲[计数]
    mov edx,[esi] / lea eax,[esp+8] / mov ecx,esi
    push eax / call [edx+0x10]   ; ★ 虚表槽 0x10（尾回调/通知）
    mov eax,1 / ret 4            ; ★ 返 1
   返 -1: or eax,-1 / ret 4

语义：**向量容器追加（Add_Item 形态，4B/项）**——检查容量（this->+0x28
上限 vs +0x20 计数，子结构 +0x18），不足时经虚表槽 8 扩展，然后 4B/项写
缓冲[计数]并计数++，最后经虚表槽 0x10 通知。
★**与 batch 167 登记的 0x6CF240（Swizzle 注册 8B/项）同族**——完全同构
的「容量检查 + 虚表槽 8 Resize + 计数++ + 写缓冲 + 虚表槽 0x10 通知」，
仅项宽 4B vs 8B；0x5D1C20 是通用 4B 向量版，0x6CF240 是 SwizzlePointer
(8B) 特化。紧邻 0x5D1C90 是同构第二成员（含删除语义，虚表槽 0x10 查找）。

⚠️ partial：宿主（0x5D1C90 起的 vector 类）、+0x28/+0x20 精确字段名、
虚表槽 8/0x10 方法身份。

0x7CA8C6  CRT 宽字符查找（11 callers，0x7CA8C6-0x7CA8EE 读完）
================================================================================
机器码（0x28 读完）：
    mov eax,[esp+4]              ; ★ 串指针
    mov cx,word[eax] / test cx / je 收尾   ; ★ 逐 2 字节扫描
    cmp cx,word[esp+8] / je 收尾          ; ★ 与目标宽字符比较
    inc eax / inc eax / jmp 开头           ; ★ 步进 2 字节
   收尾: mov cx,word[eax] / sub cx,word[esp+8]
    neg cx / sbb ecx,ecx / not ecx / and eax,ecx   ; ★ 未命中 → eax=0
    ret

语义：**宽字符逐 2 字节查找（wcslen/wcschr 形态）**——在窄串指针按 word
步进扫描直到命中目标宽字符或串尾，命中返回指针、未命中返回 NULL。
★0x7C-0x7D 区（MSVC6 静态 CRT，非 Westwood 手写——batch 169 的
0x7CA530 同区）——CRT 宽字符工具族。

⚠️ partial：宿主（0x7CA8EF 起是 SEH 帧函数——本函数为 CRT 内部工具）、
具体 CRT 函数名（wcsstr/wcschr 推断）。
