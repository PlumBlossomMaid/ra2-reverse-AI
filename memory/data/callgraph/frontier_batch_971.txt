一、链语义（disasm 全 280B 复核）
------------------------------------------------
0x7CE18D(280B) = MSVC6 CRT fputc（0x7C-0x7D 静态 CRT 区）——cdecl 2
栈参 (c, FILE*)，单字符写入带缓冲流：
  push ebp / mov ebp,esp / push ebx / push esi
  mov esi,[ebp+0xC]（FILE* 流槽）/ mov eax,[esi+0xC]（标志字）/
  mov ebx,[esi+0x10]（缓冲指针）
  test al,0x82（_IOERR|_IOEOF）→ je 0x7CE299（★无错 → 继续）
    je 不跳 → 0x7CE299: or al,0x20（置 _IOERROR）/ mov [esi+0xC],eax
      → or eax,-1（返 -1 EOF）
  test al,0x40（_IOSTRG 字符串流）→ jne 0x7CE299（★字符串流同 EOF）
  test al,1（_IOOPEN）→ je 0x7CE1C5（未打开 → 跳过复位）
    and [esi+4],0（★清读写位置）
    test al,0x10（_IOFLRTN）→ je 0x7CE299（★已 flush 标志 → EOF 分支？
      实际为条件复位）
    mov ecx,[esi+8] / and al,0xFE / mov [esi],ecx / mov [esi+0xC],eax
      （★缓冲基址 → 游标，清 _IOOPEN 位）
  0x7CE1C5（主路径）：
    mov eax,[esi+0xC] / and [esi+4],0 / and dword [ebp+0xC],0
    and al,0xEF（清 _IOFLRTN）/ or al,2（置 _IOOPEN）/ mov [esi+0xC],eax
    test ax,0x10C（_IOFLRTN|_IOBINARY）→ jne 0x7CE1FF（★需 flush/二进制）
    cmp esi,0x87BF70 → je 0x7CE1ED / cmp esi,0x87BF90 → jne 0x7CE1F8
      （★★stdout/stderr 特槽？实际为 _iob 首两槽推断未确认）
    0x7CE1ED: push ebx / call 0x7D7754（★★句柄表操作——batch 048
      0x7D7754/0x7D7710 同层）/ pop ecx / jne 0x7CE1FF（成功跳过）
    0x7CE1F8: push esi / call 0x7D7710（★★句柄表操作）/ pop ecx
  0x7CE1FF（写路径核心）：
    test word [esi+0xC],0x108（_IOFLRTN|_IOBINARY 宽判）→ je 0x7CE26F
      （非缓冲/需 flush）
    ★缓冲直写：eax=[esi+8]/edi=[esi]（游标-缓冲基）/sub/lea [eax+1]/
      mov [esi],ecx（游标+1）/mov [esi+4],ecx（[esi+0x18] 减 1 余量）
      test edi,edi → jle 0x7CE22F（★缓冲空 → 直接落字节）
      push edi/push eax/push ebx → call 0x7D0D55（★★缓冲写设施——
        batch 970 同 0x7D 段 FILE 层）/mov [ebp+0xC],eax
      0x7CE22F: cmp ebx,-1 → je 0x7CE24D（★无底层 fd）
        sar/and 索引 [ecx*4+0xB78A80]（★★句柄映射表）+ lea 组合 →
        test byte [eax+4],0x20 → je 0x7CE265
        push 2/push 0/push ebx → call 0x7D7638（★★底层写设施 fd）
      0x7CE265: mov eax,[esi+8] / mov cl,[ebp+8] / mov [eax],cl
        （★字符入缓冲）
      0x7CE283: cmp [ebp+0xC],edi → je 0x7CE28F（成功）
        or dword [esi+0xC],0x20（_IOERROR）→ or eax,-1（返 -1）
      0x7CE28F: mov eax,[ebp+8] / and eax,0xFF（★返字符低 8 位）
      0x7CE299: or al,0x20 / mov [esi+0xC],eax / or eax,-1
  收尾：pop 族 / leave / ret（cdecl 2 参弹 8B）
  0x7CE26F（非缓冲路径）：push 1/push [ebp+8]/push ebx → call 0x7D0D55
    （★单字节缓冲写）/cmp [ebp+0xC],edi → 同错误处理

★语义（batch 052 名准）：**流写单字符（fputc）**——标志字多种组合
状态机：错误位/字符串流 → EOF；打开态复位 → 缓冲直写（满则 flush）
或非缓冲单字节写 → 失败置 _IOERROR 返 -1 / 成功返字符低 8 位。
★原版怪癖（照抄）：①错误位 _IOERR/_IOEOF/_IOSTRG 任一命中即 or
0x20（_IOERROR）+ 返 -1——**错误位写回再清** 语义；②字符返回值为
低 8 位（and eax,0xFF）——高位丢弃；③缓冲空（edi<=0）时不走
0x7D0D55 而直接落字节 + 可选 0x7D7638 底层写；④句柄映射表 0xB78A80
按 (fd>>5) 行 + (fd&0x1F) 列取槽。

★依赖链实锤（本批 disasm + TSV 交叉）：
  ① 0x7D0D55/0x7D7638/0x7D7754/0x7D7710 = FILE 缓冲写/底层写/句柄
    表操作设施——batch 048/970 已交叉（0x7D 段 CRT FILE 层）；
  ② 0xB78A80 = 句柄映射表（fd 索引）——0x7CE238-0x7CE248 实锤
     (fd>>5)×4 + (fd&0x1F)×9×4 槽式计算；
  ③ 流槽标志字：0x01 _IOOPEN/0x02 _IOFLRTN 变体/0x10 _IOFLRTN/
     0x20 _IOERROR/0x40 _IOSTRG/0x80 _IOEOF/0x108 组合（batch 045/
     046/049/970 反复出现）。
